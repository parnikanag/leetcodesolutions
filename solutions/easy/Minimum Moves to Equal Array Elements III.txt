// Title: Minimum Moves to Equal Array Elements III
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/minimum-moves-to-equal-array-elements-iii/

class Solution:
    def minMoves(self, nums: list[int]) -> int:
        max_val = max(nums)
        total_sum = sum(nums)
        return max_val * len(nums) - total_sum
