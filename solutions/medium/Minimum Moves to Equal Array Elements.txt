// Title: Minimum Moves to Equal Array Elements
            // Difficulty: Medium
            // Language: Python3
            // Link: https://leetcode.com/problems/minimum-moves-to-equal-array-elements/

class Solution:
    def minMoves(self, nums: list[int]) -> int:
        min_val = min(nums)
        total_sum = sum(nums)
        
        return total_sum - len(nums) * min_val
